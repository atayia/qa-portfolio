import { test, expect } from '@playwright/test';

test.describe('TC-AMZ-003 — Product Search', () => {
  test('Search returns relevant results for a valid query', async ({ page }) => {
    await page.goto('/');
    const searchBox = page.getByRole('searchbox', { name: 'Search Amazon' });
    await searchBox.fill('Logitech MX Master 3S');
    await searchBox.press('Enter');

    await expect(page).toHaveURL(/s\?k=/);
    await expect(page.locator('.s-search-results')).toBeVisible();
    await expect(page.locator('h2.a-size-mini').first()).toBeVisible();

    // Attach the results page as evidence in the report
    await test.info().attach('Search results page', {
      body: await page.screenshot({ fullPage: true }),
      contentType: 'image/png',
    });
  });
});

Pass TC-AMZ-005 — Global Search — No Results (Negative)
Linked Test Case
TC-AMZ-005
Observation
AMZ-OBS-001 — Empty-state behavior varies by network origin
Spec File
automation/tests/amazon/search-empty.spec.ts
Notes

The first CI run in GitHub Actions marked this test flaky: it failed twice, then passed on retry. The Playwright error context showed that Amazon served a different page from the cloud runner: a "No results for your search query." heading (Variant B) instead of the "More results" sponsored fallback seen locally (Variant A).

Fix: replaced CSS and internal-attribute selectors (#twotabsearchtextbox, [data-component-type]) with Playwright's recommended role-based locators (getByRole), which target what users and screen readers perceive. The assertion accepts either variant, and each run records the variant observed as a report annotation with a full-page screenshot attached.

Result: Variant A in 5/5 local runs (residential network, fresh browser profile); Variant B in all CI runs. Finding logged to AMZ-OBS-001.

Assertions
import { test, expect } from '@playwright/test';
      
      test.describe('TC-AMZ-005 — Empty Search / No Results', () => {
        test('Non-existent query shows a no-results message or sponsored fallback', async ({ page }) => {
          await page.goto('/');
          const searchBox = page.getByRole('searchbox', { name: 'Search Amazon' });
          await searchBox.fill('zxyvwk123456789');
          await searchBox.press('Enter');
      
          await expect(page).toHaveURL(/s\?k=/);
      
          // Variant A: "More results" sponsored fallback only
          // Variant B: "No results for your search query." heading, then fallback
          const noResultsHeading = page.getByRole('heading', { name: /no results for/i });
          const moreResultsHeading = page.getByRole('heading', { name: /more results/i });
      
          await expect(noResultsHeading.or(moreResultsHeading).first()).toBeVisible();
      
          // Record which variant this run received
          const variant = (await noResultsHeading.isVisible())
            ? 'B — "No results for your search query." heading shown'
            : 'A — "More results" sponsored fallback, no empty-state message';
      
          test.info().annotations.push({ type: 'Variant observed', description: variant });
          await test.info().attach('Search results page', {
            body: await page.screenshot({ fullPage: true }),
            contentType: 'image/png',
          });
        });
      });
Result
Pass — 5/5 local (Variant A) · CI (Variant B)

Accessibility

Pass TC-AMZ-008 — Sign-In Programmatic Labels (A11y)
Linked Test Case
TC-AMZ-008
Spec File
automation/tests/amazon/signin-a11y.spec.ts
Notes

The manual test verified VoiceOver's spoken output, which automation cannot hear. This spec verifies what the screen reader reads from: the field's programmatic label. Playwright's getByLabel() only matches an input whose label is programmatically associated with it, so the test fails if the label is removed or disconnected.

Scope: email field only. Reaching the password field requires submitting credentials, which Amazon's bot detection blocks for automated sessions. Password field labelling and spoken output remain covered by the manual test.

Assertions
import { test, expect } from '@playwright/test';

test.describe('TC-AMZ-008 — Sign-In Programmatic Labels (A11y)', () => {
  test('Email field exposes an accessible name matching its visible label', async ({ page }) => {
    await page.goto('/');

    // Open the sign-in page from the header, as a user would
    await page.getByRole('link', { name: /hello, sign in/i }).click();
    await expect(page).toHaveURL(/\/ap\/signin/);

    // getByLabel finds the input through its programmatic label —
    // the same information a screen reader announces
    const emailField = page.getByLabel(/mobile number or email/i);

    await expect(emailField).toBeVisible();
    await expect(emailField).toBeEditable();

    await test.info().attach('Sign-in page', {
      body: await page.screenshot({ fullPage: true }),
      contentType: 'image/png',
    });
  });
});
