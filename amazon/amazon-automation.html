import { test, expect } from '@playwright/test';

test.describe('TC-AMZ-003 — Product Search', () => {
  test('Search returns relevant results for a valid query', async ({ page }) => {
    await page.goto('/');
    const searchBox = page.locator('#twotabsearchtextbox');
    await searchBox.fill('Logitech MX Master 3S');
    await searchBox.press('Enter');

    await expect(page).toHaveURL(/s\?k=/);
    await expect(page.locator('.s-search-results')).toBeVisible();
    await expect(page.locator('h2.a-size-mini').first()).toBeVisible();
  });
});

Pass TC-AMZ-005 — Global Search — No Results (Negative)
Linked Test Case
TC-AMZ-005
Observation
AMZ-OBS-001 — No empty-state UI; sponsored fallback returned instead
Spec File
automation/tests/amazon/search-empty.spec.ts
Notes
Amazon does not surface a true empty state for unrecognised queries. Instead, the system returns sponsored fallback products under a "More results" heading. The assertion was updated to reflect this actual behaviour — verifying that either a [data-component-type="s-no-results"] block OR a "More results" heading is present using Playwright's .or() combinator.
Assertions
import { test, expect } from '@playwright/test';

test.describe('TC-AMZ-005 — Empty Search / No Results', () => {
  test('Non-existent query returns sponsored fallback content', async ({ page }) => {
    await page.goto('/');
    const searchBox = page.locator('#twotabsearchtextbox');
    await searchBox.fill('zxyvwk123456789');
    await searchBox.press('Enter');

    await expect(page).toHaveURL(/s\?k=/);
    await expect(page.locator('.s-search-results')).toBeVisible();

    const noResultsMsg = page.locator('[data-component-type="s-no-results"]');
    const moreResultsHeading = page.locator('h2').filter({ hasText: 'More results' });

    await expect(noResultsMsg.or(moreResultsHeading)).toBeVisible();
  });
});
